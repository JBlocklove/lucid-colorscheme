%% drawing helpers
\newcommand\eyebrow[1]{{\addfontfeature{LetterSpace=12}\fontsize{7}{8}\selectfont\bfseries\MakeUppercase{#1}}}
% \ink{name}: sets the color "ink" to whichever of fg and bg differs more in
% brightness from l-name, so swatch labels read in either variant.
\newcommand\grayof[2]{\extractcolorspecs{l-#1}{\inkmodel}{\inkspec}\convertcolorspec{\inkmodel}{\inkspec}{gray}#2}
\newcommand\ink[1]{\grayof{#1}\gswatch\grayof{bg}\gbg\grayof{fg}\gfg
  \pgfmathparse{abs(\gswatch-\gbg) > abs(\gswatch-\gfg)}%
  \ifdim\pgfmathresult pt>0.5pt \colorlet{ink}{l-bg}\else\colorlet{ink}{l-fg}\fi}
% \tile{x}{y}{width}{height}{name}
\newcommand\tile[5]{%
  \ink{#5}%
  \fill[l-#5, rounded corners=2.5pt] (#1,#2) rectangle ++(#3,-#4);
  \draw[l-border, line width=0.4pt, rounded corners=2.5pt] (#1,#2) rectangle ++(#3,-#4);
  \node[anchor=south west, align=left, text=ink, inner sep=0, font=\fontsize{5.4}{6.4}\selectfont]
    at (#1+0.045,#2-#4+0.06) {\textbf{\expandafter\detokenize\expandafter{#5}}\\{\ttfamily\hex{#5}}};
}
